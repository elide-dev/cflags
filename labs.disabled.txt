# Optional flags, not loaded by either resolver. Re-enable only after
# validating the stated contract and measuring toolchain/downstream workloads.

# base.txt: permits merging distinct const objects, changing address identity.
# Ordinary string pooling does not require this non-standard optimization.
-fmerge-all-constants

# linux.txt: ABI-incompatible relative vtables. Rebuild the C++ runtime and
# every dependency exchanging polymorphic objects with the same setting.
# Compiler version alignment alone is insufficient. Test runtime virtual calls.
-fexperimental-relative-c++-abi-vtables

# base.txt: stack canaries. Benchmark startup and steady-state overhead;
# no universal percentage applies. LLVM alignment does not remove this cost.
-fstack-protector-strong

# base.txt: clear caller-saved GPRs on return. Benchmark code size and return
# paths; this is independent of CET availability.
-fzero-call-used-regs=used-gpr

# linux.txt / darwin.txt: fortified libc wrappers. Validate support in the
# actual headers: stock musl alone does not supply glibc-style fortification.
# Evaluate level 3 as well where dynamic object-size checking is supported.
-U_FORTIFY_SOURCE
-D_FORTIFY_SOURCE=2

# linux.txt / darwin.txt: libc++ fast hardening; match the vendored libc++
# configuration. Does not enable equivalent checks in libstdc++.
-D_LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST

# linux.txt: Polly is an optional build component. Benchmark affine-loop
# workloads before enabling; version alignment does not establish a benefit.
-mllvm
-polly

# linux-amd64.txt: CET instrumentation. Enforcement depends on CPU, kernel,
# loader and all linked inputs. Benchmark separately from other hardening.
-fcf-protection=full

# linux.txt: candidates now that the former JNI consumers are gone.
# Test independently with PIC shared libraries and bundled dependencies.
# -fno-semantic-interposition changes intra-DSO replacement semantics;
# -fno-plt changes external call lowering. Audit LD_PRELOAD/interposition
# requirements and benchmark both startup and steady-state behavior.
-fno-semantic-interposition
-fno-plt

# linux-amd64.txt: use only with a measured profile-driven block layout
# pipeline. Per-block sections alone can increase size and relocation work.
-fbasic-block-sections=all
