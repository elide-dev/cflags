# macOS + Apple Silicon — *final-link-only* flags. Appended after
# base.txt, darwin.txt, and darwin-arm64.txt for the consuming project's
# executable / dylib link steps. See darwin-amd64-bin.txt for the
# rationale — the same dependency-build-vs-final-link split applies on
# Apple Silicon.

# Drop unreferenced symbols/sections (Mach-O analogue of --gc-sections).
# Safe at the final link only.
-Wl,-dead_strip

# Drop unused LC_LOAD_DYLIB entries (Mach-O analogue of --as-needed).
-Wl,-dead_strip_dylibs
