# Linux x86-64 baseline, shared with elide-dev/toolchain.
# x86-64-v3 includes AVX2, BMI1/2, FMA and related instructions. Do not add
# optional crypto/GFNI/RNG extensions here: explicit -m options survive a
# downstream -march override and silently raise the deployment floor.
# Crypto libraries retain their own runtime CPU dispatch for faster paths.
-march=x86-64-v3

# Scheduling preference only; this does not require Zen 3 instructions.
-mtune=znver3
