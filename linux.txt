# Linux-only C flags — appended after cflags/base.txt for any build that
# targets Linux (linux / linux-aarch64 / linux-riscv64 profiles). Apple's
# clang either does not implement these, or the macOS runtime does not
# back them, so they have to live outside base.txt.

# Probe + clobber the unallocated stack to defeat Stack Clash attacks. clang
# implements this for Linux/{x86,x86_64,riscv} and Linux/aarch64; Apple's
# clang does not enable the codegen on darwin.
-fstack-clash-protection

# Force LLD as the linker. Required on Alpine + clang22 + ThinLTO: the
# default linker clang picks up is binutils-ld (Alpine's compiler-rt
# config), which can't read bitcode .o members from .a archives — autoconf
# conftests then fail with `member ...(foo.o) in archive is not an object`
# during the C-compiler-works check. lld handles bitcode natively. Apple's
# clang on darwin uses Apple's `ld -bitcode_bundle`-aware linker so this
# flag belongs to Linux only.
-fuse-ld=lld

# Relative C++ ABI vtables — emits a `.rtti_proxy` indirection per RTTI
# entry that the linker dedupes via COMDAT. ELF supports COMDAT (via
# section groups) so this works on Linux clang/lld; MachO does not, and
# clang aborts with `LLVM ERROR: MachO doesn't support COMDATs,
# '<symbol>.rtti_proxy' cannot be lowered` when the BoringSSL build's
# C++ test executables (decrepit_test, ssl_test, bssl, …) try to link.
# Net effect on Linux: smaller vtables + faster startup; net effect on
# darwin: cmake build aborts. Linux only.
-fexperimental-relative-c++-abi-vtables

# Skip the dynamic-symbol interposition contract for our own definitions:
# under ELF the linker assumes any global symbol in a shared library can
# be replaced via LD_PRELOAD, which forces every intra-library call through
# the GOT/PLT and disables a wide range of inlining + IPO. Disabling it
# lets clang resolve calls and constants directly inside the same DSO.
# Combined with -fno-plt below, the call-site overhead approaches a static
# build's. Linux/ELF only — Mach-O has different visibility semantics and
# clang ignores this flag there.
-fno-semantic-interposition

# Skip the PLT for external calls and route them through GOT loads instead.
# Pairs with -Wl,-z,now (full RELRO): the GOT is resolved at load time and
# made read-only, so removing the lazy-binding PLT shrinks code size,
# eliminates a layer of indirection on hot calls, and removes the writable
# PLT-stub gadget surface from the binary. Linux/ELF only.
-fno-plt

# Glibc fortified-source: replace selected libc calls (str*, mem*, sprintf
# family, read/write, …) with __*_chk variants that bounds-check at runtime
# whenever the compiler can constant-fold the destination size. Level 3
# extends the analysis to non-constant sizes via __builtin_dynamic_object_size
# and requires glibc 2.34+ / clang ≥ 9 / -O1+ (we use -O3). On musl it
# silently degrades to level 2, which is still strictly better than
# unfortified. Defining it here also shadows any weaker -D_FORTIFY_SOURCE
# the upstream Makefiles may set.
-U_FORTIFY_SOURCE
-D_FORTIFY_SOURCE=3

# libstdc++ runtime assertions: turn on bounds checks in std::vector,
# std::string, std::span, iterators, etc. Cheap (mostly inlined) and
# catches indexing/iterator misuse in the BoringSSL C++ test binaries.
# No-op when libstdc++ headers aren't included. Linux only because Apple
# ships libc++ — see darwin.txt for the libc++ analogue.
-D_GLIBCXX_ASSERTIONS

# Pass through to the LLVM backend and enable Polly, the polyhedral loop
# optimizer. Performs high-level transformations (tiling, fusion, skewing,
# auto-parallelization, improved vectorization) on affine loop nests by
# modeling them as polyhedra. Requires a Clang built with Polly support
# (the official LLVM clang-22 build does); gains are concentrated in
# numeric/array-heavy code and are negligible or negative elsewhere.
-mllvm
-polly

# Mark assembler-produced objects as not requiring an executable stack.
# Without this, a single .s/.S input lacking a .note.GNU-stack section
# poisons the linker into emitting PT_GNU_STACK as RWX for the whole binary,
# silently undoing -Wl,-z,noexecstack.
-Wa,--noexecstack

# Linker hardening + size flags:
#   relro                  - mark GOT and other ELF sections read-only after relocation.
#   now                    - disable lazy PLT binding; resolve all symbols at load time.
#                            Combined with relro this yields full RELRO, hardening the
#                            GOT against overwrite (and it's required to make -fno-plt
#                            useful, since there is no longer a lazy resolver to fall
#                            back to).
#   noexecstack            - set PT_GNU_STACK to non-executable; blocks shellcode
#                            execution from the stack.
#   separate-code          - place executable code in its own pages with no readable
#                            data alongside, reducing info leaks and limiting ROP
#                            gadget exposure.
#   pack-relative-relocs   - emit DT_RELR-format relative relocations instead of the
#                            full RELA encoding. Substantially smaller .rela.dyn for
#                            PIE/PIC binaries on glibc 2.36+ / lld 15+.
#   --gc-sections          - drop unreferenced sections produced by -ffunction-sections
#                            / -fdata-sections (this is the consumer of those flags).
#   --as-needed            - skip DT_NEEDED entries for shared libs whose symbols are
#                            never actually referenced; trims runtime dep surface.
#   --icf=safe             - lld identical-code-folding, "safe" variant: only fold
#                            functions whose address isn't taken. Frees code size
#                            without breaking function-pointer identity.
#   --build-id=sha1        - embed a deterministic build ID for crash-server lookup.
#   -O2                    - linker-side string/section optimization (small win, free).
-Wl,-z,relro
-Wl,-z,now
-Wl,-z,noexecstack
-Wl,-z,separate-code
-Wl,-z,pack-relative-relocs
-Wl,--gc-sections
-Wl,--as-needed
-Wl,--icf=safe
-Wl,--build-id=sha1
-Wl,-O2
